81/*
82 * The PLL clock frequency is calculated the following way:
83 *
84 * pllfreq = (multiplier * basefreq) * pllTable[pllOutSelect],
85 *
86 * with basefreq = 37 MHz and multiplier = 9 (by default)
87 *
88 * <pllOutSelect> is an index into a table defining values which will yield
89 * the desired clock frequency value when inserted into the formula above.
90 *
91 * Example: Setting the PLL's out select value to ::SCE_SYSREG_PLL_OUT_SELECT_222MHz
92 * the value at pllTable[SCE_SYSREG_PLL_OUT_SELECT_222MHz] will be used in the
93 * PLL frequency calculation, thus we get:
94 *
95 * pllfreq = (37 MHz * 9) * 0.667 = ~222 MHz
96 *
97 * Remarks:
98 *
99 * Note that as of at least 6.60, pllTable[pllOutSelect] now
100 * returns the final clock frequency:
101 *
102 * pllfreq = pllTable[pllOutSelect] MHz
103 *
104 * In other words,
105 * pllTable[SCE_SYSREG_PLL_OUT_SELECT_222MHz] now returns 222 instead of 0.667.
106 *
107 * That table can be found in the lowio module at address 0x0000C1C0. It's also
108 * worth pointing out that the PLL clock frequencies 37 MHz and 148 MHz don't seem
109 * to be available any longer. If we study the pllTable content, we see that it starts
110 * with 190 MHz all the way up to the 166 MHz entry (in other words, each index now is
111 * appears to be index - 2 internally).
112 */
113#define SCE_SYSREG_PLL_OUT_SELECT_37MHz (0)
114#define SCE_SYSREG_PLL_OUT_SELECT_148MHz (1)
115#define SCE_SYSREG_PLL_OUT_SELECT_190MHz (2)
