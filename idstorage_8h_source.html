38/*
39 * ID Storage layout (WIP):
40 *
41 * Each ID storage leaf has a size of 512 bytes. Some leafs are "self-contained", that is, they do not share data with
42 * their neighboring leafs. Examples for such leafs are the leafs with IDs 0x4 - 0x8. Other leafs are "continious" leafs,
43 * meaning they share data with their neighboring leafs. Examples of such leafs are the leafs 0x100 - 0x102.
44 *
45 *
46 * .
47 * .
48 * .
49 *
50 * The leafs 0x100 - 0x102 contain the ConsoleId (termed "PSID" on the PSP) and the OpenPSID. As an additional security feature
51 * to prevent successfully tampering them, they are stored inside a certificate, which can be used to verify the ConsoleId and
52 * OpenPSID. For more info, see SceIdStorageConsoleIdCertificate and SceIdStorageOpenPSIDCertificate.
53 *
54 * These leafs are "continious leafs" and contain 5 ConsoleId certificates (totalling 5 * 184 = 920 bytes)
55 * followed by a single OpenPSID certificate (for a total size of 920 + 184 = 1104 bytes).
56 *
57 * Of these 5 certificates, only the first ConsoleId certificate and the OpenPSID certificate are used in the system.
58 * It is unknown what the purpose the remaining four ConsoleId certificates is.
59 *
60 * +------------------------------------------------------------+----------------
61 * | unknown data (size: 0x38) | |
62 * |------------------------------------------------------------| |
63 * | SceIdStorageConsoleIdCertificate (size: 0xB8) | |
64 * |------------------------------------------------------------| Leaf 0x100
65 * | SceIdStorageConsoleIdCertificate (size: 0xB8) | |
66 * |------------------------------------------------------------| |
67 * | SceIdStorageConsoleIdCertificate_part1 (size: 0x58) | |
68 * |------------------------------------------------------------|----------------
69 * | SceIdStorageConsoleIdCertificate_part2 (size: 0x60) | |
70 * |------------------------------------------------------------| |
71 * | SceIdStorageConsoleIdCertificate (size: 0xB8) | |
72 * |------------------------------------------------------------| Leaf 0x101
73 * | SceIdStorageConsoleIdCertificate (size: 0xB8) | |
74 * |------------------------------------------------------------| |
75 * | SceIdStorageOpenPSIDCertificate_part1 (size: 0x30) | |
76 * |------------------------------------------------------------|----------------
77 * | SceIdStorageOpenPSIDCertificate_part2 (size: 0x88) | |
78 * |------------------------------------------------------------| Leaf 0x102
79 * | unknown data (size: 0x178) | |
80 * |------------------------------------------------------------|----------------
81 * .
82 * .
83 * .
84 *
85 * The leafs 0x120 - 0x126 are backups of leafs 0x100 - 0x106. See the description of those leafs for more information.
86 *
87 * .
88 * .
89 * .
90 *
91 */

93/*
94 * This structure contains the ConsoleId (termed "PSID" on the PSP) and an ECDSA signature used to verify the correctness of the
95 * ConsoleId.
96 * The ConsoleId is used, for example, in PSN DRM, DNAS and system configuration (with its derived PSCode).
97 */
98typedef struct {
99 /* Unique per-console identifier. */
100 SceConsoleId consoleId; // 0
101 /* Contains the public key of the certificate. No padding. */
102 u8 plantextPublicKey[0x28]; // 16
103 /* The 'r' part of the ECDSA signature pair (r, s). */
104 u8 r[0x14]; // 56
105 /* The 's' part of the ECDSA signature pair (r, s). */
106 u8 s[0x14]; // 76
107 /* The ECDSA public key (can be used to verify ECDSA signature rs). */
108 u8 publicKey[0x28]; // 96
109 /* Contains the encrypted private key of the certificate (with padding). */
110 u8 encPrivateKey[0x20]; // 136
111 /* Hash of previous data. */
112 u8 hash[0x10]; // 168
113} SceIdStorageConsoleIdCertificate; // size = 184

115/*
116 * This structure contains the OpenPSID (random 16 byte value) and an ECDSA signature used to verify the correctness
117 * of the OpenPSID. The OpenPSID value is used, for example, in PSN matchmaking.
118 */
119typedef struct {
120 /* Unique per-console identifier. */
121 u8 openPSID[0x10]; // 0
122 /* Contains the public key of the certificate. No padding. */
123 u8 plantextPublicKey[0x28]; // 16
124 /* The 'r' part of the ECDSA signature pair (r, s). */
125 u8 r[0x14]; // 56
126 /* The 's' part of the ECDSA signature pair (r, s). */
127 u8 s[0x14]; // 76
128 /* The ECDSA public key (can be used to verify ECDSA signature rs). */
129 u8 publicKey[0x28]; // 96
130 /* Contains the encrypted private key of the certificate (with padding). */
131 u8 encPrivateKey[0x20]; // 136
132 /* Hash of previous data. */
133 u8 hash[0x10]; // 168
134} SceIdStorageOpenPSIDCertificate; // size = 184
